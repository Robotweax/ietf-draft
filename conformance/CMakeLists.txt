cmake_minimum_required(VERSION 3.20)
project(srt_draft_conformance LANGUAGES CXX)
set(SRT_SOURCE "" CACHE PATH "Immutable exported Robotweax SRT 0.3.2 source")
set(FIXTURE_HEADER_DIR "" CACHE PATH "Directory containing generated fixtures.hpp")
if(NOT EXISTS "${SRT_SOURCE}/include/robotweax/srt/crypto.hpp" OR NOT EXISTS "${FIXTURE_HEADER_DIR}/fixtures.hpp")
  message(FATAL_ERROR "Set SRT_SOURCE and FIXTURE_HEADER_DIR")
endif()
set(ROBOTWEAX_SRT_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(ROBOTWEAX_SRT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ROBOTWEAX_SRT_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(ENABLE_AEAD_API_PREVIEW ON CACHE BOOL "" FORCE)
add_subdirectory("${SRT_SOURCE}" srt)
add_executable(srt_draft_adapter adapter.cpp admission_handoff.cpp)
target_compile_features(srt_draft_adapter PRIVATE cxx_std_20)
target_include_directories(srt_draft_adapter PRIVATE "${SRT_SOURCE}/include" "${SRT_SOURCE}/src" "${FIXTURE_HEADER_DIR}")
target_link_libraries(srt_draft_adapter PRIVATE robotweax_srt)
