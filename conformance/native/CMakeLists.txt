cmake_minimum_required(VERSION 3.20)
project(srt_native_publication_probe LANGUAGES CXX)
set(ROBOTWEAX_SRT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ROBOTWEAX_SRT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ROBOTWEAX_SRT_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(ENABLE_AEAD_API_PREVIEW ON CACHE BOOL "" FORCE)
add_subdirectory("${SRT_SOURCE}" srt)
add_executable(native_probe native_probe.cpp)
target_compile_features(native_probe PRIVATE cxx_std_20)
target_include_directories(native_probe PRIVATE "${SRT_SOURCE}/src" "${SRT_SOURCE}/include")
target_link_libraries(native_probe PRIVATE robotweax_srt)
